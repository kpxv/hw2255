\subsection{Theory: Circuit Investigated}

Calculation of $R_\text{F.INV}$:
\begin{align*}
	\frac{V_{in}}{R_1}             & = - \frac{V_{out}}{R_f} \\
	-\frac{R_f}{R_1}               & = \frac{V_{out}}{V_in}  \\
	-\frac{R_f}{\left(1000\right)} & = \left(-4.7\right)     \\
	R_f                            & = 4700
\end{align*}

Calculation of $R_\text{F.NON\_INV}$:
\begin{align*}
	-\frac{V_{in}}{R_1}        & = \frac{V_{out} - V_{in}}{R_f} \\
	\frac{V_{in}}{V_{out}} - 1 & = \frac{R_1}{R_f}              \\
	\frac{V_{out}}{V_{in}}     & = \frac{R_f}{R_1} + 1          \\
	3.2                        & = \frac{R_f}{1000} + 1         \\
	R_f                        & = 2200
\end{align*}
